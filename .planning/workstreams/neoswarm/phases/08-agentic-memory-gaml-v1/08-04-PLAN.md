```cpp
/**
 * @file       memory_governor.hpp
 * @brief      Standalone memory retrieval orchestrator (GAML v1 — Phase 8)
 */
#ifndef NEOSWARM_MEMORY_MEMORYGOVERNOR_HPP
#define NEOSWARM_MEMORY_MEMORYGOVERNOR_HPP

#include "common/error.hpp"
#include "common/types.hpp"
#include <memory>
#include <string>

namespace sgns::neoswarm::memory
{
    class MemoryStorage;  // forward declaration — avoid circular include

    /**
     * @brief Retrieves relevant memory objects using heuristic entity-match +
     *        recency + confidence ranking.
     *
     * Phase 8 (v1): Heuristic only. ML-assisted retrieval deferred.
     *
     * Does NOT implement IELM — standalone orchestration component (D-18).
     * Owned by ApiServer (D-04). Called between prompt analysis and RunELMChain.
     */
    class MemoryGovernor
    {
        public:
        struct Config
        {
            int m_topK = 10;                                  ///< max results returned
            float m_minConfidence = 0.3f;                      ///< minimum confidence to include
            int64_t m_maxAgeNs = 86400LL * 1000000000LL;       ///< 24h default age cutoff
            bool m_enabled = true;
        };

        /// @param storage Non-owning pointer — owned by ApiServer per D-04
        explicit MemoryGovernor(Config cfg, MemoryStorage* storage);
        ~MemoryGovernor();

        /// @return True if governor is ready to serve queries.
        bool IsReady() const { return m_storage != nullptr && m_cfg.m_enabled; }

        /**
         * @brief Retrieve memory context for a task (D-05, D-06).
         *
         * 1. Extract entity from task.m_prompt (keyword heuristic)
         * 2. GetByPrefix on entity for facts + policies
         * 3. Filter by recency, rank by confidence-weighted recency
         * 4. Return top-k results as MemoryContext
         *
         * @param task   Incoming task (used for entity extraction).
         * @param chain  Execution chain (used for context building).
         * @return       MemoryContext with facts + policies.
         *               Empty context on no results — NOT an error (D-20).
         */
        MemoryContext Retrieve(const Task& task, const ExecutionChain& chain) const;

        private:
        Config m_cfg;
        MemoryStorage* m_storage;  ///< non-owning — owned by ApiServer

        std::string ExtractEntity(const std::string& prompt) const;
        std::vector<CognitiveAsset> FilterAndRank(
            const std::vector<CognitiveAsset>& candidates) const;
    };

} // namespace sgns::neoswarm::memory

#endif // NEOSWARM_MEMORY_MEMORYGOVERNOR_HPP
```

4. **ExtractEntity()** — keyword-based heuristic (D-06, OpenCode's Discretion):
```cpp
std::string MemoryGovernor::ExtractEntity(const std::string& prompt) const
{
    std::string lower = prompt;
    std::transform(lower.begin(), lower.end(), lower.begin(),
                   [](unsigned char c) { return std::tolower(c); });

    // Common question/request prefixes to strip
    static const std::vector<std::string> kPrefixes = {
        "explain", "what is", "tell me about", "describe", "define",
        "how does", "why is", "who is", "where is", "when did"
    };

    for (const auto& prefix : kPrefixes)
    {
        auto pos = lower.find(prefix);
        if (pos != std::string::npos && pos + prefix.size() < lower.size())
        {
            std::string rest = prompt.substr(pos + prefix.size());
            std::istringstream iss(rest);
            std::string word, entity;
            int count = 0;
            while (iss >> word && count < 3)  // take first 3 words as entity
            {
                if (!entity.empty()) entity += "_";
                entity += word;
                ++count;
            }
            return entity;
        }
    }
    // Fallback: first 60 chars of prompt as entity
    return prompt.substr(0, 60);
}
```

5. **FilterAndRank()** — recency-weighted confidence ranking (D-06):
```cpp
std::vector<CognitiveAsset> MemoryGovernor::FilterAndRank(
    const std::vector<CognitiveAsset>& candidates) const
{
    std::vector<std::pair<float, size_t>> scored;
    scored.reserve(candidates.size());

    int64_t now = std::chrono::steady_clock::now().time_since_epoch().count();
    int64_t cutoff = now - m_cfg.m_maxAgeNs;

    for (size_t i = 0; i < candidates.size(); ++i)
    {
        const auto& obj = candidates[i];

        // D-06: filter by confidence threshold
        if (obj.m_confidence < m_cfg.m_minConfidence) continue;

        // D-06: filter by recency
        if (obj.m_timestamp < cutoff) continue;

        // Recency-weighted confidence: newer = bonus
        float ageRatio = static_cast<float>(obj.m_timestamp - cutoff) /
                         static_cast<float>(m_cfg.m_maxAgeNs);
        float score = obj.m_confidence * (0.7f + 0.3f * ageRatio);
        scored.push_back({score, i});
    }

    // Sort descending by score
    std::sort(scored.begin(), scored.end(),
              [](const auto& a, const auto& b) { return a.first > b.first; });

    // Take top-k
    std::vector<CognitiveAsset> results;
    for (size_t i = 0; i < scored.size() &&
         static_cast<int>(results.size()) < m_cfg.m_topK; ++i)
    {
        results.push_back(candidates[scored[i].second]);
    }
    return results;
}
```
**Note on scoring formula:** `confidence * (0.7 + 0.3 * ageRatio)` — newer facts get up to 30% boost over stale ones. This is OpenCode's Discretion.

6. **Retrieve()** — the main orchestration method (D-05):
```cpp
MemoryContext MemoryGovernor::Retrieve(const Task& task, const ExecutionChain& chain) const
{
    MemoryContext ctx;

    // D-20: disabled or no storage → empty context (not error)
    if (!IsReady() || !m_storage || !m_storage->IsOpen())
    {
        GovernorLogger()->debug("MemoryGovernor not ready — returning empty context");
        return ctx;
    }

    // Step 1: Extract entity
    std::string entity = ExtractEntity(task.m_prompt);
    GovernorLogger()->debug("Extracted entity: '{}'", entity);

    // Step 2: Dual GetByPrefix — facts + policies (D-06)
    std::string factPrefix = entity + "/" +
        std::to_string(static_cast<int>(MemoryObjectType::fact)) + "/";
    std::string policyPrefix = entity + "/" +
        std::to_string(static_cast<int>(MemoryObjectType::policy)) + "/";

    auto factsRes = m_storage->GetByPrefix(factPrefix, m_cfg.m_topK * 2);
    auto policiesRes = m_storage->GetByPrefix(policyPrefix, m_cfg.m_topK * 2);

    // Step 3: Filter + rank
    if (factsRes.has_value())
        ctx.m_facts = FilterAndRank(factsRes.value());
    if (policiesRes.has_value())
        ctx.m_policies = FilterAndRank(policiesRes.value());

    GovernorLogger()->info("Retrieved {} facts, {} policies for entity '{}'",
                           ctx.m_facts.size(), ctx.m_policies.size(), entity);
    return ctx;
}
```
**Note:** Retrieves topK*2 candidates then filters → ensures after confidence filtering we still have up to topK results. Entity + type prefix format matches BuildKey from 08-02.

**Error handling (D-19):** `GetByPrefix` returns are checked via `.has_value()`. When storage is disabled/unavailable, returns empty MemoryContext — no error propagation needed (D-20). The `ExecutionChain` parameter is accepted but not used in v1 (heuristic retrieval doesn't need chain context — ML-assisted would).

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User prompt → ExtractEntity | Entity name is derived from user-supplied prompt text — could be manipulated to query unintended entity prefixes |
| Retrieve → MemoryStorage::GetByPrefix | Prefix construction uses entity string directly — must sanitize `/` (handled by MemoryStorage::BuildKey) |
| FilterAndRank → CognitiveAsset timestamps | Timestamps could be in the future (clock skew) or zero (default) — ageRatio calculation must handle edge cases |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-13 | Information Disclosure | ExtractEntity prompt parsing | accept | Entity extraction uses only the first 3 words after common prefixes — scope of information leakage is limited to prefix-matching entity names. Full access control deferred to Phase 10 |
| T-08-14 | Denial of Service | GetByPrefix large result sets | mitigate | topK*2 limits maximum results. GetByPrefix itself has maxResults parameter. RocksDB iterator bounds prevent unbounded scans |
| T-08-15 | Elevation of Privilege | Entity prefix injection from prompt | mitigate | Entity names are sanitized by MemoryStorage::BuildKey (replaces `/` → `_`). The `/` delimiter cannot be injected through ExtractEntity's word-based extraction |
| T-08-16 | Denial of Service | FilterAndRank division by zero | mitigate | `m_maxAgeNs` is non-zero (default 24h ns). `ageRatio` numerator checks bounds: `(obj.m_timestamp - cutoff) / m_maxAgeNs` where cutoff = now - maxAgeNs. If timestamp in future, ageRatio > 1.0 which is clamped implicitly by the multiplication factor |
</threat_model>
