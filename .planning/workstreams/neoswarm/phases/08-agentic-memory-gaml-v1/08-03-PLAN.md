```cpp
static const std::regex kSentencePattern(R"(([^.!?]+[.!?]))");  // split by sentence boundaries
auto begin = std::sregex_iterator(text.begin(), text.end(), kSentencePattern);
int64_t now = std::chrono::steady_clock::now().time_since_epoch().count();
for (auto it = begin; it != end && static_cast<int>(facts.size()) < m_cfg.m_maxFacts; ++it)
{
    std::string sentence = it->str();
    // Trim whitespace
    sentence.erase(0, sentence.find_first_not_of(" \t\n\r"));
    sentence.erase(sentence.find_last_not_of(" \t\n\r") + 1);
    if (sentence.empty() || sentence.size() < 10) continue;
    if (!IsDeclarativeSentence(sentence)) continue;
    float confidence = ScoreFactConfidence(sentence);
    if (confidence < m_cfg.m_minConfidence) continue;

    CognitiveAsset fact;
    fact.m_id = "mem-" + std::to_string(now) + "-" + std::to_string(facts.size());
    fact.m_entity = "";  // set by ContextMapping in Stage 2
    fact.m_type = MemoryObjectType::fact;      // D-08: all extracted are facts
    fact.m_payload = nlohmann::json::object({{"content", sentence}, {"source_task_id", task.m_id}});
    fact.m_timestamp = now;
    fact.m_sourceNode = task.m_nodeId;
    fact.m_confidence = confidence;
    fact.m_provenance = 0.5f;
    fact.m_trustClass = TrustClass::unverified;  // D-09: privacy stub
    facts.push_back(std::move(fact));
}
return outcome::success(std::move(facts));
```

- **Evaluate() method:**
```cpp
outcome::result<std::vector<CognitiveAsset>> WriteEvaluation::Evaluate(
    std::vector<CognitiveAsset> facts, MemoryStorage* storage) const
{
    if (!m_cfg.m_enabled) return outcome::success(std::move(facts));  // D-07: can disable

    std::vector<CognitiveAsset> accepted;
    accepted.reserve(facts.size());

    for (auto& fact : facts)
    {
        float novelty = ScoreNovelty(fact, storage);
        float utility = ScoreUtility(fact);

        if (novelty < m_cfg.m_noveltyThreshold) continue;
        if (utility < m_cfg.m_utilityThreshold) continue;

        float overall = (novelty + utility) / 2.0f;
        if (overall < m_cfg.m_overallThreshold) continue;

        fact.m_confidence = overall;  // update confidence with evaluation score
        accepted.push_back(std::move(fact));
    }
    EvalLogger()->debug("Write evaluation: {} accepted / {} total", accepted.size(), facts.size());
    return outcome::success(std::move(accepted));
}
```

- **ScoreNovelty()** — check if similar entity+type facts exist in storage:
```cpp
float WriteEvaluation::ScoreNovelty(const CognitiveAsset& fact, MemoryStorage* storage) const
{
    if (!storage || !storage->IsOpen()) return 0.5f;  // can't check → neutral

    std::string prefix = fact.m_entity + "/" +
        std::to_string(static_cast<int>(fact.m_type)) + "/";
    auto existing = storage->GetByPrefix(prefix, 1);  // just check existence
    if (!existing.has_value() || existing.value().empty()) return 0.8f;  // nothing similar = novel
    return 0.2f;  // similar exists = less novel
}
```

- **ScoreUtility()** — length + keyword heuristic:
  - Base: 0.3f
  - Medium-length content (50-300 char): +0.3f
  - Long content (300-800 char): +0.1f
  - Specificity keywords (`is`, `means`, `defined`, `approximately`, `measured`, `discovered`, `introduced`, `published`, `synthesized`): +0.05 each
  - Clamped to [0.0, 1.0]
  - Extracts `payload["content"]` string for analysis

**Per D-20:** Failed write evaluation → `InternalError` (but the Evaluate method itself doesn't produce that — it returns accepted facts or error only if storage calls fail). Novelty fails gracefully with neutral score if storage unavailable.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| ELM output text → FactExtraction | Untrusted inference output passes through regex parsing; malformed text could trigger regex catastrophic backtracking |
| External entity names → ContextMapping | Entity names derived from user prompts flow into RocksDB keys |
| WriteEvaluation scoring → MemoryStorage | Novelty checks call GetByPrefix — storage unavailability handled gracefully |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-09 | Denial of Service | FactExtraction std::regex backtracking | mitigate | Sentence-splitting regex `([^.!?]+[.!?])` has bounded backtracking (character class negation). m_maxFacts limits iterations. Configurable m_minConfidence filters noise |
| T-08-10 | Information Disclosure | ContextMapping entity extraction | accept | Entity names derived from prompt text — only user-facing content, no secrets. Phase 10 privacy classification will add encryption |
| T-08-11 | Tampering | WriteEvaluation novelty bypass | accept | Novelty score returns 0.5f default when storage unavailable — attacker can't exploit this to suppress fact ingestion in Phase 8 (local-only) |
| T-08-12 | Repudiation | Ingestion pipeline → MemoryStorage | accept | Phase 8 is local-only; no multi-node provenance tracking. Full attestation with cryptographic signatures deferred to Phase 9 |
</threat_model>
