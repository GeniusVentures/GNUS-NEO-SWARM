#include <gtest/gtest.h>
#include <string>

using namespace sgns::neoswarm::security;

namespace
{
    // Helper: convert NodeIdentity::PubKey to hex string for use as pub_key_hex
    std::string PubKeyToHex( const NodeIdentity::PubKey &pubKey )
    {
        std::ostringstream oss;
        for ( auto b : pubKey )
        {
            oss << std::hex << std::setw( 2 ) << std::setfill( '0' )
                << static_cast<int>( b );
        }
        return oss.str();
    }
}

TEST( MessageSigningTest, VerifyAndStripRoundtrip )
{

    ASSERT_TRUE( id.Generate().has_value() );

    MessageSigning signer( id );
    std::string    original = "{\"action\":\"chat\",\"prompt\":\"hi\"}";
    std::string    signedPayload = signer.AttachSignature( original );

    std::string pubKeyHex = PubKeyToHex( id.PublicKey() );
    // VerifyAndStrip modifies the payload in-place
    bool        valid     = MessageSigning::VerifyAndStrip( signedPayload, pubKeyHex );
    ASSERT_TRUE( valid );

    // After stripping, we should have recovered the original (or close to it)
    // The original should be a substring (minus injected nonce/ts fields)
    ASSERT_NE( signedPayload.find( "\"action\":\"chat\"" ), std::string::npos );
    ASSERT_NE( signedPayload.find( "\"prompt\":\"hi\"" ), std::string::npos );

    // sig field should be gone
    ASSERT_EQ( signedPayload.find( "\"sig\":" ), std::string::npos );
}
