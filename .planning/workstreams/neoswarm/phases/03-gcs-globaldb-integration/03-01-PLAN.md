<objective>
Create the GCS GlobalDB component skeleton: a new `src/storage/` module owning a NEO-SWARM-side `sgns::crdt::GlobalDB` with an init-style lifecycle. The component pulls the shared GossipPubSub from the in-process GeniusNode (via `GeniusSDKGetNode()->GetPubSub()`), constructs io_context/scheduler/graphsync network/request-id generator locally, calls `GlobalDB::New` with `datastore = nullptr`, starts the database, and wires the `gcs-reputation` CRDT listen+broadcast topic.

Purpose: Establish the storage foundation that Phase 3 (reputation), Phase 8 (`/gcs/memory/`), and Phase 9 (`/gcs/consensus/`) all build on — one NEO-SWARM-owned GlobalDB, separate from the blockchain GlobalDB inside GeniusSDK (D-01).

Output: `neoswarm_storage` CMake target with a compilable, lifecycle-tested `GcsGlobalDb` component. No consumers rewired yet.
</objective>

class GlobalDB : public std::enable_shared_from_this<GlobalDB>
{
    struct BackupOptions { bool enabled{false}; uint32_t interval_minutes{15};
                           uint32_t keep_count{12}; bool auto_restore_on_repair_failure{true}; };
    using Buffer = base::Buffer;

    static outcome::result<std::shared_ptr<GlobalDB>> New(
        std::shared_ptr<boost::asio::io_context>                              context,
        std::string                                                           databasePath,
        std::shared_ptr<sgns::ipfs_pubsub::GossipPubSub>                      pubsub,
        std::shared_ptr<CrdtOptions>                                          crdtOptions,
        std::shared_ptr<sgns::ipfs_lite::ipfs::graphsync::Network>            graphsyncnetwork,
        std::shared_ptr<libp2p::basic::Scheduler>                             scheduler,
        std::shared_ptr<sgns::ipfs_lite::ipfs::graphsync::RequestIdGenerator> generator,
        std::shared_ptr<RocksDB>                                              datastore = nullptr,
        BackupOptions backup_options = BackupOptions{ false, 15, 12, true } );

    enum class Error : uint8_t
    {
        ROCKSDB_IO = 0, IPFS_DB_NOT_CREATED, DAG_SYNCHER_NOT_LISTENING,
        CRDT_DATASTORE_NOT_CREATED, PUBSUB_BROADCASTER_NOT_CREATED,
        INVALID_PARAMETERS, GLOBALDB_NOT_STARTED,
    };

    outcome::result<void> AddBroadcastTopic( const std::string &topicName );
    void                  AddListenTopic( const std::string &topicName );
    void Start();
    void ShutdownNow();   // quiesce CRDT intake + workers; safe to call multiple times
};
